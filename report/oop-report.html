Numerical Method Calculator — React + Express + SQLite

รายงานฉบับนี้อธิบายการออกแบบเชิงวัตถุ (OOP Design) ที่ใช้จริงในโค้ดของโปรเจกต์ FastNumeric โดยยกตัวอย่างจากไฟล์ source code จริงทุกตัวอย่าง

1แนะนำโปรเจกต์

FastNumeric เป็นเว็บแอปพลิเคชันสำหรับคำนวณโจทย์ Numerical Methods ครบทั้ง 6 หมวด (Root of Equation, Linear Algebra, Interpolation, Extrapolation, Integration, Differentiation) รวม 11 วิธีคำนวณ พร้อมระบบบันทึกประวัติการคำนวณลงฐานข้อมูล SQLite และรันได้ทั้งแบบปกติและผ่าน Docker

โจทย์หลักของการออกแบบคือ “เพิ่มวิธีคำนวณใหม่ได้โดยไม่ต้องแก้ระบบเดิม” — ทั้งระบบจึงถูกออกแบบรอบคลาสนามธรรม (Abstract Class) ชื่อ BaseMethod โดยทุกวิธีคำนวณเป็นคลาสลูกที่ extend ออกมา

แผนภาพความสัมพันธ์ของคลาสในระบบอัลกอริทึม (สร้างจากโค้ดจริงใน server/src/algorithms/)

BaseMethod ทำหน้าที่เป็น สัญญา (contract) ของทุกวิธีคำนวณ: ทุกคลาสลูกต้องมีข้อมูลประกอบ (meta) ที่บอกว่าใช้ field อะไรบ้าง และต้อง implement เมธอด solve() ที่รับ input มาตรฐานชุดเดียวกัน และคืนผลลัพธ์รูปแบบเดียวกันเสมอ

server/src/algorithms/BaseMethod.js — Abstract Class ของทุกวิธีคำนวณ 
class BaseMethod {
  constructor(meta) {
    if (!meta || !meta.key || !meta.category || !meta.name) {
      throw new Error('method meta ต้องมี key, category และ name');
    }
    this.meta = meta;
  }

  /** @abstract */
  solve(_input) {
    throw new Error(`วิธี '${this.meta.key}' ยังไม่ได้ implement เมธอด solve()`);
  }
}

4แนวคิด OOP ที่ประยุกต์ใช้ในโปรเจกต์

4.1 Abstraction — สัญญาเดียวสำหรับทุกวิธีคำนวณ

BaseMethod กำหนด ว่าทุกวิธีต้องทำอะไร (มี meta, มี solve) โดยไม่สนใจ วิธีที่จะทำ ทำให้ฝั่ง API เขียนโค้ดกับ “ความนามธรรม” ได้เลย:

server/src/routes/index.js — API คำนวณ (ไม่รู้จักอัลกอริทึมใดเลย) 
router.post('/calculate', (req, res) => {
  const method = getMethod(methodKey);        // ได้ BaseMethod ตัวใดตัวหนึ่ง
  const parsed = validateInput(method, input); // ตรวจตาม field ของวิธีนั้น
  const result = method.solve(parsed);         // ← สั่งผ่านสัญญาเดียว
  history.saveCalculation({ ...result });
  res.json({ result });
});

class BisectionMethod extends BaseMethod {

จุดที่น่าสังเกตคือ fields ใน meta ทำหน้าที่สองอย่างพร้อมกัน: ฝั่ง API ใช้เป็น สเปกตรวจ input อัตโนมัติ และฝั่ง frontend ใช้ สร้างฟอร์มไดนามิก — เพิ่มวิธีใหม่แล้วฟอร์มกับ validation เกิดขึ้นเองโดยไม่ต้องเขียนเพิ่ม

4.3 Polymorphism — อินเตอร์เฟซเดียว พฤติกรรมต่างกัน

การคำนวณครั้งหนึ่งอาจเป็น Bisection, Gauss หรือ Simpson ก็ได้ — โค้ด API ชุดเดียวทำงานได้กับทั้ง 11 วิธี เพราะทุกคลาสตอบสนองข้อความ solve() เดียวกันด้วยอัลกอริทึมของตัวเอง

ตัวอย่างที่ชัดที่สุดคือ DifferenceMethod — คลาสเดียวสร้าง 3 วิธี (Forward, Backward, Central) ซึ่งต่างกันแค่สูตรและเงื่อนไขจุดปลายตาราง โดยเลือกพฤติกรรมผ่าน constructor:

server/src/algorithms/differentiation.js — คลาสเดียว 3 พฤติกรรม

4.4 Encapsulation — ซ่อนข้อมูลดิบ ให้เห็นแต่ข้อมูลที่ผ่านการตรวจแล้ว

4.5 Composition — Registry ประกอบร่างทุกวิธีเข้าด้วยกัน

registry.js เป็นตัวประกอบ (composer) ที่รวมอ็อบเจกต์วิธีคำนวณทั้งหมดไว้ใน Map เดียว การเพิ่มวิธีใหม่ต้องแก้แค่ไฟล์นี้ไฟล์เดียว (Open/Closed Principle):

server/src/algorithms/registry.js — จุดเดียวที่ต้องแก้เมื่อเพิ่มวิธีใหม่ 
const methods = new Map();

function register(method) {
  if (methods.has(method.meta.key)) {
    throw new Error(`method key ซ้ำ: ${method.meta.key}`);
  }
  methods.set(method.meta.key, method);
}

[
  new BisectionMethod(),
  new NewtonRaphsonMethod(),
  new GaussEliminationMethod(),
  new LinearInterpolationMethod(),
  new LagrangeInterpolationMethod(),
  new LinearExtrapolationMethod(),
  new TrapezoidalRuleMethod(),
  new SimpsonsOneThirdRuleMethod(),
  new DifferenceMethod('forward'),
  new DifferenceMethod('backward'),
  new DifferenceMethod('central'),

และ GET /api/methods อ่าน meta จาก registry นี้เพื่อบอก frontend ว่าแต่ละวิธีต้องกรอกอะไรบ้าง — UI, validation และอัลกอริทึมมาจากแหล่งความจริงจุดเดียวกัน (Single Source of Truth)

5ตัวอย่างอัลกอริทึมที่ถูกที่สุด

ตัวอย่างต่อไปนี้คือ Bisection Method ฉบับเต็มจากโปรเจกต์ — จะเห็นว่าแต่ละรอบของการวนซ้ำถูกเก็บลง rows เพื่อสร้างตาราง iteration และตรวจเงื่อนไขก่อนทำงานเสมอ (ห้าม f(XL)·f(XU) > 0):

server/src/algorithms/roots.js — solve() ของ Bisection (ย่อส่วนผลลัพธ์) 
solve({ fx, xl, xu, tol, maxIter }) {
  if (xl >= xu) throw badRequest('ค่า XL ต้องน้อยกว่า XU');

  const fxl = evalAt(fx, xl, 'f(XL)');
  const fxu = evalAt(fx, xu, 'f(XU)');
  if (fxl * fxu > 0) {
    throw badRequest('f(XL) และ f(XU) มีเครื่องหมายเดียวกัน — ไม่มีการเปลี่ยน' +
                     'เครื่องหมายในช่วงนี้ จึงไม่สามารถใช้ Bisection ได้');
  }

  const rows = [];
  let a = xl, b = xu, prev = null, iter = 0;
  let root = (a + b) / 2;

  for (; iter < maxIter; iter++) {
    const fa = iter === 0 ? fxl : evalAt(fx, a, 'f(XL)');
    root = (a + b) / 2;                      // XR = (XL + XU) / 2
    const froot = evalAt(fx, root, 'f(x)');
    const error = prev === null ? '-' : Math.abs((root - prev) / root) * 100;

    rows.push({ iter: iter + 1, xl: a, xu: b, xr: root,
                fxl: fa, fxr: froot, error });   // เก็บไว้ทำตาราง

    if (froot === 0) break;
    if (fa * froot < 0) b = root; else a = root; // เลือกครึ่งช่วงใหม่
    if (prev !== null && error < tol) break;     // ลู่เข้าแล้ว
    prev = root;
  }

  return {
    answer: `รากของสมการ (root) ≈ ${fmt(root)}`,
    answerValue: root,
    steps: [ /* สูตร, เงื่อนไข, การทำซ้ำ */ ],
    table: { columns: ['#','XL','XU','XR','f(XL)','f(XR)','Error (%)'],
             rows: rows.map(...) },
    meta: { iterations: rows.length, converged: iter < maxIter },
  };
}

6สรุป

จุดขายของการออกแบบนี้อยู่ที่การทำให้ “ความรู้เรื่องวิธีคำนวณ” กระจุกอยู่ในคลาสของมันเองเท่านั้น — ทั้งฟอร์มที่ผู้ใช้กรอก, การตรวจ input, ตาราง iteration ที่แสดงผล และประวัติที่บันทึกลงฐานข้อมูล ล้วนถูกขับเคลื่อนจาก meta และ solve() ของคลาสแต่ละตัวโดยอัตโนมัติ

เมื่อต้องเพิ่มวิธีใหม่ (เช่น Secant Method หรือ Gauss-Seidel) งานที่ต้องทำคือสร้างคลาสใหม่ extends BaseMethod หนึ่งคลาส แล้วเพิ่ม register() หนึ่งบรรทัด — ฟอร์ม, validation, ผลลัพธ์, ขั้นตอนวิธีทำ และการบันทึกประวัติทั้งหมดจะทำงานกับวิธีใหม่ทันที โดยที่ระบบเดิมไม่ถูกแตะต้องแม้แต่บรรทัดเดียว นี่คือพลังของ Abstraction, Inheritance และ Polymorphism ที่ทำงานร่วมกันอย่างเป็นระบบ
